tokens.css 复制
/* ── Color ── */ :root { --color-brand: #FFB800; --color-brand-dim: #F0A500; --color-danger: #F03030; --color-surface: #16161C; --color-surface-2: #1E1E26; --color-text: #FFFFFF; --color-text-2: #A0A0A8; /* ── Spacing ── */ --space-1: 4px; --space-2: 8px; --space-4: 16px; --space-8: 32px; /* ── Radius ── */ --radius-sm: 4px; --radius-md: 8px; --radius-lg: 12px; --radius-xl: 16px; }
